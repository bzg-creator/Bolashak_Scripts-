SELECT 'Cards.Country',
       c.CountryDictionaryId,
       c.CountryDictionaryValue,
       m.NewId,
       m.NewValue,
       COUNT(*)
FROM dbo.Cards c
JOIN #CountryMap m
  ON UPPER(LTRIM(RTRIM(c.CountryDictionaryId))) = UPPER(m.OldId)
GROUP BY c.CountryDictionaryId, c.CountryDictionaryValue, m.NewId, m.NewValue

UNION ALL

SELECT 'Cards.Country (NULL Id)',
       c.CountryDictionaryId,
       c.CountryDictionaryValue,
       m.NewId,
       m.NewValue,
       COUNT(*)
FROM dbo.Cards c
JOIN #CountryMap m
  ON LTRIM(RTRIM(c.CountryDictionaryValue)) = m.OldValue
WHERE c.CountryDictionaryId IS NULL OR LTRIM(RTRIM(c.CountryDictionaryId)) = ''
GROUP BY c.CountryDictionaryId, c.CountryDictionaryValue, m.NewId, m.NewValue

ORDER BY 1, 6 DESC;

-- Cards.Country
UPDATE c
SET c.CountryDictionaryId = m.NewId,
    c.CountryDictionaryValue = m.NewValue
FROM dbo.Cards c
JOIN #CountryMap m
  ON UPPER(LTRIM(RTRIM(c.CountryDictionaryId))) = UPPER(m.OldId)
WHERE m.OldId IS NOT NULL;
