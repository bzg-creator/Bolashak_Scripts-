-- ============================================================
-- Обновление языков для записей без Iin
-- Ищем по старым Id и Value
-- ============================================================

-- ШАГ 1 — АНАЛИЗ
SELECT
    Iin,
    UniversityRkLanguageDictionaryId    AS OldId,
    UniversityRkLanguageDictionaryValue AS OldValue
FROM dbo.Cards
WHERE UniversityRkLanguageDictionaryId IN (
    '516fff8e-09d8-4ab8-a738-b90ae746935a',
    '958590f2-730d-4d56-9927-e450f4e3096e',
    '573f2c5b-5360-4437-a2c9-908ca31e1396',
    '17857e99-9b37-4407-8f95-ac5819d1c4a2'
)
AND (Iin IS NULL OR LTRIM(RTRIM(Iin)) = '');
